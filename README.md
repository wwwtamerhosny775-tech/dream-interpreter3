        function interpretDream() {
            const input = document.getElementById('dreamInput');
            const text = input.value.trim();

            if (text.length < 2) {
                alert('⚠️ الرجاء إدخال وصف أوضح للحلم.');
                return;
            }

            let found = null;
            let keyword = null;

            for (const [key, value] of Object.entries(dreamDB)) {
                if (text.includes(key)) {
                    found = value;
                    keyword = key;
                    break;
                }
            }

            let result;
            if (found) {
                result = {
                    موجود: true,
                    تفسير: found.تفسير,
                    نفسي: found.نفسي,
                    روحي: found.روحي,
                    نصيحة: found.نصيحة,
                    مفتاح: keyword
                };
            } else {
                const msg = kindMessages[Math.floor(Math.random() * kindMessages.length)];
                result = {
                    موجود: false,
                    تفسير: msg,
                    نفسي: "لا يمكن تحديده بدقة، لكنه يعكس مشاعرك الحالية.",
                    روحي: "قد يكون إشارة من روحك، استمع إليها.",
                    نصيحة: "استغفر الله وتوكل عليه، واطلب الرؤية الصالحة."
                };
            }

            displayResult(result);
        }
